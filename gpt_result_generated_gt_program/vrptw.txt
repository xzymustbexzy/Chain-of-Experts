要解决带时间窗口的车辆路径问题 (VRPTW)，通常需要执行复杂的优化步骤，可能涉及启发式算法、元启发式算法或混合整数线性规划（MILP）技术等。在这里，我们可以使用Python中的MILP解决方案，比如Google的OR-Tools或其他如PuLP的库来建模和解决问题。

以下是使用Google的OR-Tools解决此问题的框架代码示例。请注意，由于这是一个复杂问题，并且可能需要大量前期准备工作，以下代码需要在已经正确安装并配置OR-Tools的系统上运行。

```python
from ortools.constraint_solver import pywrapcp
from ortools.constraint_solver import routing_enums_pb2

def vrptw(customers, demand, lower_time_window, upper_time_window, distance, service_time, vehicle_capacity, M):
    num_vehicles = len(customers)
    depot = 0
    
    # Create the routing index manager.
    manager = pywrapcp.RoutingIndexManager(len(distance), num_vehicles, depot)
    
    # Create Routing Model.
    routing_model = pywrapcp.RoutingModel(manager)

    # Create and register a transit callback.
    def distance_callback(from_index, to_index):
        from_node = manager.IndexToNode(from_index)
        to_node = manager.IndexToNode(to_index)
        return distance[from_node][to_node]
    
    transit_callback_index = routing_model.RegisterTransitCallback(distance_callback)
    
    # Set the cost of travel.
    routing_model.SetArcCostEvaluatorOfAllVehicles(transit_callback_index)

    # Add Capacity constraint.
    def demand_callback(from_index):
        from_node = manager.IndexToNode(from_index)
        return demand[from_node]
    demand_callback_index = routing_model.RegisterUnaryTransitCallback(demand_callback)
    routing_model.AddDimensionWithVehicleCapacity(
        demand_callback_index,
        0,  # null capacity slack
        [vehicle_capacity] * num_vehicles,  # vehicle maximum capacities
        True,  # start cumul to zero
        'Capacity')

    # Add Time Window constraint.
    time = 'Time'
    routing_model.AddDimension(
        transit_callback_index,
        30,  # allow waiting time
        30,  # maximum time per vehicle
        False,  # Don't force start cumul to zero.
        time)
    time_dimension = routing_model.GetDimensionOrDie(time)
    for customer_id in range(1, len(customers)):
        start = lower_time_window[customer_id]
        end = upper_time_window[customer_id]
        index = manager.NodeToIndex(customer_id)
        time_dimension.CumulVar(index).SetRange(start, end)

    # Setting first solution heuristic.
    search_parameters = pywrapcp.DefaultRoutingSearchParameters()
    search_parameters.first_solution_strategy = (
        routing_enums_pb2.FirstSolutionStrategy.PATH_CHEAPEST_ARC)
    search_parameters.local_search_metaheuristic = (
        routing_enums_pb2.LocalSearchMetaheuristic.GUIDED_LOCAL_SEARCH)
    search_parameters.time_limit.FromSeconds(1)
    
    # Solve the problem.
    solution = routing_model.SolveWithParameters(search_parameters)
    if solution:
        print_solution(manager, routing_model, solution)
    
    # Return the minimum total distance.
    min_total_distance = solution.ObjectiveValue()
    return min_total_distance

def print_solution(manager, routing, solution):
    print("Objective:", solution.ObjectiveValue())
    time_dimension = routing.GetDimensionOrDie('Time')
    total_distance = 0
    for vehicle_id in range(num_vehicles):
        index = routing.Start(vehicle_id)
        print(f'Route for vehicle {vehicle_id}:')
        while not routing.IsEnd(index):
            time_var = time_dimension.CumulVar(index)
            print(f'{manager.IndexToNode(index)} Time({solution.Min(time_var)}, {solution.Max(time_var)}) -> ', end='')
            previous_index = index
            index = solution.Value(routing.NextVar(index))
            total_distance += routing.GetArcCostForVehicle(previous_index, index, vehicle_id)
        time_var = time_dimension.CumulVar(index)
        print(f'{manager.IndexToNode(index)} Time({solution.Min(time_var)}, {solution.Max(time_var)})')
    print("Total distance:", total_distance)

# Customers definition
# assuming customers[0] is the depot
# Example of input
customers = [i for i in range(5)]  # for example, 5 customers
demand = [0, 10, 15, 10, 20]  # demand at each customer
lower_time_window = [0, 5, 10, 15, 20]
upper_time_window = [10, 15, 20, 25, 30]
distance = [[0, 2, 9, 10, 15],
            [1, 0, 6, 7, 12],
            [3, 4, 0, 1, 6],
            [10, 8, 2, 0, 3],
            [7, 6, 1, 2, 0]]
service_time = [0, 2, 2, 2, 2]
vehicle_capacity = 50
M = 1000  # Big enough

# Solve the problem
min_total_distance = vrptw(customers, demand, lower_time_window, upper_time_window, distance, service_time, vehicle_capacity, M)

print("Minimum total distance:", min_total_distance)
```
